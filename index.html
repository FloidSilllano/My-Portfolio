// Select DOM elements const checkInForm = document.getElementById('checkInForm'); const attendeeNameInput = document.getElementById('attendeeName'); const teamSelect = document.getElementById('teamSelect'); const greeting = document.getElementById('greeting'); const attendeeCountSpan = document.getElementById('attendeeCount'); const progressBar = document.getElementById('progressBar'); const waterCountSpan = document.getElementById('waterCount'); const zeroCountSpan = document.getElementById('zeroCount'); const powerCountSpan = document.getElementById('powerCount'); // Initialize tracking variables let totalAttendees = 0; let waterCount = 0; let zeroCount = 0; let powerCount = 0; const maxCapacity = 50; // Defined goal from index.html // Listen for form submission checkInForm.addEventListener('submit', function (event) { // Prevent page reload on submit event.preventDefault(); // Get input values const name = attendeeNameInput.value.trim(); const team = teamSelect.value; // Validate inputs if (name === '' || team === '') return; // Increment total attendance totalAttendees++; // Increment specific team count and update variable if (team === 'water') { waterCount++; waterCountSpan.textContent = waterCount; } else if (team === 'zero') { zeroCount++; zeroCountSpan.textContent = zeroCount; } else if (team === 'power') { powerCount++; powerCountSpan.textContent = powerCount; } // Update total attendance display attendeeCountSpan.textContent = totalAttendees; // Update progress bar width percentage const progressPercentage = (totalAttendees / maxCapacity) * 100; progressBar.style.width = progressPercentage + '%'; // Display personalized greeting greeting.textContent = `Welcome, ${name}! Successfully checked in for Team ${getTeamDisplayName(team)}.`; // Clear form inputs for the next entry attendeeNameInput.value = ''; teamSelect.selectedIndex = 0; }); // Helper function to get clean team display names function getTeamDisplayName(teamKey) { if (teamKey === 'water') return 'Water Wise'; if (teamKey === 'zero') return 'Net Zero'; if (teamKey === 'power') return 'Renewables'; return ''; }
